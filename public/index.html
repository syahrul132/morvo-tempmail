Dokumentasi API Singkat

// Contoh penggunaan API MORVO TempMail

// 1. Buat mailbox baru
POST /api/v1/mailbox
Headers: { "Authorization": "Bearer YOUR_API_KEY" }
Body: {}

// 2. Ambil daftar pesan
GET /api/v1/mailbox/{id}/messages
Headers: { "Authorization": "Bearer YOUR_API_KEY" }

// 3. Baca pesan
GET /api/v1/messages/{id}
Headers: { "Authorization": "Bearer YOUR_API_KEY" }

// 4. Kirim email
POST /api/v1/send
Headers: { "Authorization": "Bearer YOUR_API_KEY" }
Body: {
  "from": "your@mailbox.id",
  "to": "tujuan@contoh.com",
  "subject": "Hello",
  "body": "Pesan Anda"
}
